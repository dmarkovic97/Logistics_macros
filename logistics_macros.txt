Sub ProcessSAPRows()
    Dim SapGuiAuto As Object
    Dim application As Object
    Dim connection As Object
    Dim session As Object
    Dim CurrentRow As Long
    Dim lastRow As Long
    Dim ValueD As String
    Dim ValueG As String
    Dim ValueC As String
    Dim LastFourCharsC As String
    Dim Route As String
    Dim FirstFourChars As String
    Dim ValueE As Variant

    ' Establish SAP connection
    On Error Resume Next
    Set SapGuiAuto = GetObject("SAPGUI")
    If SapGuiAuto Is Nothing Then
        MsgBox "SAP GUI is not running. Please start SAP and try again.", vbExclamation
        Exit Sub
    End If

    Set application = SapGuiAuto.GetScriptingEngine
    If application Is Nothing Then
        MsgBox "Unable to access SAP scripting engine. Ensure scripting is enabled.", vbExclamation
        Exit Sub
    End If

    Set connection = application.Children(0)
    If connection Is Nothing Then
        MsgBox "No active SAP connection found.", vbExclamation
        Exit Sub
    End If

    Set session = connection.Children(0)
    If session Is Nothing Then
        MsgBox "No active SAP session found.", vbExclamation
        Exit Sub
    End If
    On Error GoTo 0

    ' Get the last row in the sheet (assuming column A is filled)
    lastRow = Worksheets("Sheet3").Cells(Worksheets("Sheet1").Rows.Count, 1).End(xlUp).row

    ' Loop through rows to find the first row with no value in column H
    For CurrentRow = 1 To lastRow
        If IsEmpty(Worksheets("Sheet3").Cells(CurrentRow, 8).Value) Then ' Column H = 8
            ' Check if column E has the value 28 or 26
            ValueE = Worksheets("Sheet3").Cells(CurrentRow, 5).Value ' Column E = 5
            If ValueE = 28 Or ValueE = 26 Then
                ' Get values from the specified columns
                ValueD = Worksheets("Sheet3").Cells(CurrentRow, 4).Value ' Column D = 4
                ValueG = Worksheets("Sheet3").Cells(CurrentRow, 7).Value ' Column G = 7
                ValueC = Worksheets("Sheet3").Cells(CurrentRow, 3).Value ' Column C = 3

                ' Extract the first 4 characters and last 4 characters
                FirstFourChars = Left(ValueC, 4)
                LastFourCharsC = Right(ValueC, 4)

                ' Determine the route based on the value of FirstFourChars
                If StrComp(FirstFourChars, "RS10", vbTextCompare) = 0 Then
                    Route = "J25261"
                ElseIf StrComp(FirstFourChars, "RS70", vbTextCompare) = 0 Then
                    Route = "J25269"
                Else
                    Route = "" ' Default or error case
                End If

                ' Perform SAP automation
                Call SAPAutomation(session, FirstFourChars, ValueD, ValueG, LastFourCharsC, Route)

                ' Exit the loop after processing the first valid row
                Exit For
            End If
        End If
    Next CurrentRow
End Sub

Sub SaveAsAndExit(filePath As String)

    
    ' Check if the file exists and delete it
    If Dir(filePath) <> "" Then
        Kill filePath
    End If
    
    ' Save the file
    ActiveWorkbook.SaveAs fileName:=filePath, FileFormat:=xlOpenXMLWorkbook
    
    ' Introduce a delay (e.g., 5 seconds)

    
    ' Close the workbook
    ActiveWorkbook.Close SaveChanges:=False
    
    ' Close Excel
    application.Quit
    
    ' Allow the system to process the quit command
    DoEvents
End Sub
